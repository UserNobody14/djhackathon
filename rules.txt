type Transition = {
  trackA: {
    id: string; // Spotify ID of Track A
    bpm: number; // Beats Per Minute
    key: string; // Musical key (e.g., C Major)
    energy: number; // Energy level (0.0 to 1.0)
    outro: {
      start: number; // Timestamp (in seconds) where the outro starts
      end: number; // Timestamp (in seconds) where the outro ends
    };
  };
  trackB: {
    id: string; // Spotify ID of Track B
    bpm: number; // Beats Per Minute
    key: string; // Musical key (e.g., G Major)
    energy: number; // Energy level (0.0 to 1.0)
    intro: {
      start: number; // Timestamp (in seconds) where the intro starts
      end: number; // Timestamp (in seconds) where the intro ends
    };
  };
  transition: {
    bpmAlignment: {
      originalBPM: {
        trackA: number;
        trackB: number;
      };
      adjustedBPM: number; // If tempo alignment is applied
    };
    keyCompatibility: {
      compatible: boolean; // Whether the keys are compatible
      adjustmentRequired: boolean; // If filters or effects are needed
    };
    crossfade: {
      duration: number; // Duration of the crossfade (in seconds)
      start: number; // Timestamp where the crossfade starts
      end: number; // Timestamp where the crossfade ends
    };
    energyFlow: {
      fadeOut: number; // Adjusted energy for fade-out
      fadeIn: number; // Adjusted energy for fade-in
    };
    effects: {
      keyFilterApplied: boolean; // Whether key correction filters are applied
      reverb: {
        applied: boolean; // Whether reverb is used
        settings?: {
          decay: number; // Reverb decay time
          mix: number; // Reverb mix percentage
        };
      };
      equalizer: {
        adjustments: Array<{
          band: string; // Frequency band (e.g., "bass", "mid", "treble")
          gain: number; // Gain adjustment in dB
        }>;
      };
    };
  };
